# Exercise backend-specific openPMD cases and shared I/O/H5hut round trips.
function(ippl_io_test executable name backend ranks)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runs/${name}")
  if(ranks EQUAL 1)
    add_test(NAME ${name} COMMAND $<TARGET_FILE:${executable}>)
  else()
    add_test(NAME ${name} COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks}
      ${MPIEXEC_PREFLAGS} $<TARGET_FILE:${executable}> ${MPIEXEC_POSTFLAGS})
  endif()
  set(_env "IPPL_TEST_IO_BACKEND=${backend}")
  if(backend STREQUAL "bp5")
    list(APPEND _env "IPPL_TEST_BP5=1")
  elseif(backend MATCHES "^h5hut")
    list(APPEND _env "IPPL_TEST_H5HUT=1")
  endif()
  set_tests_properties(${name} PROPERTIES TIMEOUT 180 PROCESSORS ${ranks}
    LABELS "unit;io;${backend}" ENVIRONMENT "${_env}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runs/${name}")
endfunction()

if(IPPL_ENABLE_OPENPMD)
  add_ippl_test(OpenPMDRoundTrip COMPILE_ONLY)
  if("HDF5" IN_LIST IPPL_OPENPMD_BACKENDS)
    ippl_io_test(OpenPMDRoundTrip OpenPMDRoundTrip hdf5 2)
    ippl_io_test(OpenPMDRoundTrip OpenPMDRoundTripSerial hdf5 1)
  endif()
  if("ADIOS2" IN_LIST IPPL_OPENPMD_BACKENDS)
    ippl_io_test(OpenPMDRoundTrip OpenPMDRoundTripBP5 bp5 2)
    ippl_io_test(OpenPMDRoundTrip OpenPMDRoundTripSerialBP5 bp5 1)
  endif()
endif()
add_ippl_test(IORoundTrip COMPILE_ONLY)
if(IPPL_ENABLE_ALPINE)
  add_ippl_test(LandauDampingStreamTest COMPILE_ONLY
    INCLUDE_DIRS "${PROJECT_SOURCE_DIR}/demos/alpineStream")
endif()
foreach(backend IN ITEMS hdf5 bp5 h5hut)
  if((backend STREQUAL "hdf5" AND IPPL_ENABLE_OPENPMD AND "HDF5" IN_LIST IPPL_OPENPMD_BACKENDS)
     OR (backend STREQUAL "bp5" AND IPPL_ENABLE_OPENPMD AND "ADIOS2" IN_LIST IPPL_OPENPMD_BACKENDS)
     OR (backend STREQUAL "h5hut" AND IPPL_ENABLE_H5HUT))
    foreach(ranks IN ITEMS 1 2)
      ippl_io_test(IORoundTrip IORoundTrip_${backend}_${ranks} ${backend} ${ranks})
    endforeach()
    if(IPPL_ENABLE_ALPINE)
      if(backend STREQUAL "hdf5")
        set(_name LandauDampingStreamTest)
      elseif(backend STREQUAL "bp5")
        set(_name LandauDampingStreamTestBP5)
      else()
        set(_name LandauDampingStreamTestH5hut)
      endif()
      ippl_io_test(LandauDampingStreamTest ${_name} ${backend} 2)
    endif()
  endif()
endforeach()
